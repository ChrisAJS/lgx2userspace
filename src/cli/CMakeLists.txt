cmake_minimum_required(VERSION 3.18)

set(SOURCES main.cpp OptionParser.cpp OptionParser.h)
set(GLFW_SOURCES glfwmain.cpp OptionParser.cpp OptionParser.h)

if(CMAKE_SYSTEM MATCHES Windows)
    message("Building Windows executable")
    add_executable(lgx2userspace WIN32 ${SOURCES} appicon.rc)
    target_include_directories(lgx2userspace PRIVATE liblgx)
    target_link_libraries(lgx2userspace PRIVATE mingw32 liblgx)
    target_link_options(lgx2userspace PRIVATE -static)

    add_executable(lgx2userspace-glfw WIN32 ${GLFW_SOURCES} appicon.rc)
    target_include_directories(lgx2userspace-glfw PRIVATE liblgx)
    target_link_libraries(lgx2userspace-glfw PRIVATE mingw32 liblgx)
    target_link_options(lgx2userspace-glfw PRIVATE -static)
else()
    find_library(SYSTEMD_LIB systemd)
    find_library(ASYNCNS_LIB asyncns)

    add_executable(lgx2userspace ${SOURCES})
    target_include_directories(lgx2userspace PRIVATE liblgx)
    target_link_libraries(lgx2userspace PRIVATE liblgx)
    if(SYSTEMD_LIB)
        target_link_libraries(lgx2userspace PRIVATE ${SYSTEMD_LIB})
    endif()
    if(ASYNCNS_LIB)
        target_link_libraries(lgx2userspace PRIVATE ${ASYNCNS_LIB})
    endif()

    add_executable(lgx2userspace-glfw ${GLFW_SOURCES})
    target_include_directories(lgx2userspace-glfw PRIVATE liblgx)
    target_link_libraries(lgx2userspace-glfw PRIVATE liblgx)
    if(SYSTEMD_LIB)
        target_link_libraries(lgx2userspace-glfw PRIVATE ${SYSTEMD_LIB})
    endif()
    if(ASYNCNS_LIB)
        target_link_libraries(lgx2userspace-glfw PRIVATE ${ASYNCNS_LIB})
    endif()
endif()
